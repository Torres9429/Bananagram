// Los privilegios viven en role_permissions (BD), nunca hardcodeados en
// código (ver CLAUDE.md). module.slug + action.slug + allowed.
model Role {
  id          String           @id @default(uuid())
  name        String           @unique
  users       UserRole[]
  permissions RolePermission[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@map("roles")
}

// Multi-rol (2026-08-05): tabla puente que reemplaza el antiguo
// User.roleId escalar — un usuario puede tener N roles a la vez (ej. un
// usuario que opera como community_manager y disenador). Los permisos
// efectivos de un usuario son la UNIÓN de los permisos de TODOS sus roles
// (ver AuthRepository.getPermissions). No se puede dejar a un usuario con
// 0 roles (AuthRepository.removeRoleFromUser lo bloquea).
model UserRole {
  userId String
  roleId String
  user   User @relation(fields: [userId], references: [id], onDelete: Cascade)
  role   Role @relation(fields: [roleId], references: [id], onDelete: Cascade)

  @@id([userId, roleId])
  @@map("user_roles")
}

model Module {
  id          String           @id @default(uuid())
  slug        String           @unique
  name        String
  permissions RolePermission[]

  @@map("modules")
}

model Action {
  id          String           @id @default(uuid())
  slug        String           @unique
  name        String
  permissions RolePermission[]

  @@map("actions")
}

model RolePermission {
  id       String  @id @default(uuid())
  roleId   String
  role     Role    @relation(fields: [roleId], references: [id])
  moduleId String
  module   Module  @relation(fields: [moduleId], references: [id])
  actionId String
  action   Action  @relation(fields: [actionId], references: [id])
  allowed  Boolean @default(true)

  @@unique([roleId, moduleId, actionId])
  @@map("role_permissions")
}

// familyId/revokedAt agregados 2026-07-27 (migración
// auth-service/prisma/migrations/20260727143453_refresh_token_family):
// todos los tokens nacidos de la misma sesión (login/register + cada
// rotación de refresh) comparten familyId. AuthRepository.rotateRefreshToken
// consume el token viejo y crea el reemplazo en una sola transacción; si
// alguien reusa un token ya consumido/revocado/expirado, se revoca TODA la
// familia (incluido el token que "ganó" la rotación) — detección de robo de
// refresh token, no solo rechazo del intento inválido.
model RefreshToken {
  id        String    @id @default(uuid())
  userId    String
  user      User      @relation(fields: [userId], references: [id])
  token     String    @unique
  familyId  String
  expiresAt DateTime
  usedAt    DateTime?
  revokedAt DateTime?
  createdAt DateTime  @default(now())

  @@index([familyId])
  @@map("refresh_tokens")
}

// Bandeja de notificaciones (todos los roles). El evento que dispara una
// notificación nace en core-service (post rechazado, campaña asignada,
// reporte listo), pero la tabla vive aquí porque el dato es userId-céntrico.
// core-service llama a un endpoint interno de auth-service (a construir) al
// ocurrir el evento — llamada irreducible, el evento no puede "reubicarse":
// nace donde nace.
model Notification {
  id        String    @id @default(uuid())
  userId    String
  user      User      @relation(fields: [userId], references: [id])
  type      String
  payload   Json
  readAt    DateTime?
  createdAt DateTime  @default(now())

  @@map("notifications")
}

// Respaldo en Postgres de la denylist de access tokens revocados por

// Inmutable — instancia LOCAL de auditoría de auth-service. performedBy se
// puebla directo desde el userId del JWT (dato que ya se tiene en cada

  tableName   String
  recordId    String

  after       Json?

  requestId   String?

  createdAt DateTime @default(now())

  @@map("audit_log")
}

// CORE
// Nota (implementación real, 2026-08-01): sin `user User @relation(...)` — no
// es posible, `User` vive en la BD de auth-service y Prisma no soporta
// `@relation` cruzando datasources. `userId` es una FK "tonta" más, igual que
// ownerId/cmId/createdBy en el resto de este archivo.
model UserProfile {
  id            String   @id @default(uuid())
  userId        String   @unique // FK → auth-service.User.id (sin constraint de BD)

  name          String
  avatarUrl     String?

  // Denormalizado desde auth-service.User.roles (multi-rol, 2026-08-05) al
  // registrar/completar perfil (POST /internal/user-profiles) — valores de
  // 'cliente'|'cm'|'disenador'. Sin esto, core-service no tiene forma de
  // saber los roles de un usuario para validar que un cmId/designerId de
  // Campaign corresponda a un CM/Diseñador real (ver
  // CampaignsService.assertUserHasRole). Array (antes un solo String?)
  // porque un usuario puede tener varios roles a la vez — @default([]) en
  // vez de null, sigue siendo best-effort como el resto del perfil.
  roleNames     String[] @default([])

  categories    UserProfileCategory[]
  specialties   UserProfileSpecialty[]

  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  deletedAt     DateTime?

  @@map("user_profiles")
}
