// ════════════════════════════════════════════════════════════════════════
// modelo2.txt — modelo.txt separado por microservicio
// ════════════════════════════════════════════════════════════════════════
//
// Punto de partida: docs/base/modelo.txt (schema Prisma único, 18+ modelos,
// el modelo original/maestro completo — NO se toca, se mantiene intacto a
// propósito como referencia). Este archivo es la propuesta de reparto por
// microservicio, y por eso SÍ puede divergir de modelo.txt (p. ej. agrega
// ContentIdea, que no existe en el maestro) — la idea es poder comparar
// "original vs. propuesta" entre ambos archivos.
//
// La organización final del backend es de 3 microservicios:
//   - auth-service   — login y todo lo derivado de autenticación
//   - core-service   — "tiene todo": marcas, redes sociales, campañas,
//                       publicaciones, medios, métricas, score, reportes
//                       e ideas de contenido (Alexa Skill)
//   - alexa-service  — SIN base de datos propia. Es un puro consumidor del
//                       API de core-service y auth-service (BFF para la
//                       Alexa Skill) — por eso NO tiene sección de schema
//                       en este archivo, no es dueño de ninguna tabla. Ver
//                       docs/base/service-boundaries.md para el detalle de
//                       qué endpoints consume de cada uno.
//
// Solo auth-service y core-service tienen datasource propio (2 secciones
// abajo). Donde antes brands/content/analytics-service se llamaban entre sí
// por HTTP (p. ej. analytics leyendo posts de content, o content validando
// una cuenta social de brands), esas llamadas cruzadas DESAPARECEN: ahora
// es el mismo servicio y la misma base de datos, así que son joins/queries
// normales de Prisma. Lo único que sigue cruzando de verdad es la identidad
// (auth-service ↔ core-service) — ver leyenda de anotaciones abajo.
//
// Cada sección trae su propio bloque `datasource`/`generator` (con su
// propia variable de entorno *_URL) como si ya fuera un archivo de schema
// independiente — así, separar físicamente en
// apps/backend/services/<servicio>/prisma/schema.prisma más adelante es
// copiar y pegar la sección completa, no rediseñarla.
//
// ── Leyenda de anotaciones en campos cruzados ──────────────────────────
//
// 1) `// FK → auth-service.<Modelo>.id (sin constraint de BD; resolver vía
//    HTTP GET /internal/..., circuit breaker)`
//    → Es una FK "tonta": una columna con el id de un registro que vive en
//      auth-service. No hay `@relation` de Prisma (no se puede: son bases
//      de datos distintas). Para escribir, casi siempre basta el id que ya
//      se tiene (del JWT o del payload del request) — no requiere llamada.
//      Para leer/mostrar datos del otro lado (nombre, email, etc.), sí
//      requiere una llamada HTTP síncrona a través del circuit breaker de
//      apps/backend/commons/circuit-breaker/opossum.factory.ts (hoy
//      implementado pero sin usar en ningún servicio).
//
// AuditLog se repite igual (mismo shape) en las 2 secciones: cada servicio
// escribe su propia tabla local de auditoría, poblada directo desde el
// `userId` del JWT — no requiere ninguna llamada cruzada para escribir.
//
// ════════════════════════════════════════════════════════════════════════

// Los privilegios viven en role_permissions (BD), nunca hardcodeados en
// código (ver CLAUDE.md). module.slug + action.slug + allowed.
model Role {
  id          String           @id @default(uuid())
  name        String           @unique
  users       User[]
  permissions RolePermission[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@map("roles")
}

model Module {
  id          String           @id @default(uuid())
  slug        String           @unique
  name        String
  permissions RolePermission[]

  @@map("modules")
}

model Action {
  id          String           @id @default(uuid())
  slug        String           @unique
  name        String
  permissions RolePermission[]

  @@map("actions")
}

model RolePermission {
  id       String  @id @default(uuid())
  roleId   String
  role     Role    @relation(fields: [roleId], references: [id])
  moduleId String
  module   Module  @relation(fields: [moduleId], references: [id])
  actionId String
  action   Action  @relation(fields: [actionId], references: [id])
  allowed  Boolean @default(true)

  @@unique([roleId, moduleId, actionId])
  @@map("role_permissions")
}

model RefreshToken {
  id        String    @id @default(uuid())
  userId    String
  user      User      @relation(fields: [userId], references: [id])
  token     String    @unique
  expiresAt DateTime
  usedAt    DateTime?
  createdAt DateTime  @default(now())

  @@map("refresh_tokens")
}

// Bandeja de notificaciones (todos los roles). El evento que dispara una
// notificación nace en core-service (post rechazado, campaña asignada,
// reporte listo), pero la tabla vive aquí porque el dato es userId-céntrico.
// core-service llama a un endpoint interno de auth-service (a construir) al
// ocurrir el evento — llamada irreducible, el evento no puede "reubicarse":
// nace donde nace.
model Notification {
  id        String    @id @default(uuid())
  userId    String
  user      User      @relation(fields: [userId], references: [id])
  type      String
  payload   Json
  readAt    DateTime?
  createdAt DateTime  @default(now())

  @@map("notifications")
}

model User {
  id                  String    @id @default(uuid())
  email               String    @unique
  passwordHash        String

model PasswordResetToken {
  id        String    @id @default(uuid())
  userId    String
  user      User      @relation(fields: [userId], references: [id])
  token     String    @unique
  expiresAt DateTime
  usedAt    DateTime?
  createdAt DateTime  @default(now())

  @@map("password_reset_tokens")
}

// Inmutable — instancia LOCAL de auditoría de auth-service. performedBy se
// puebla directo desde el userId del JWT (dato que ya se tiene en cada
// request), sin ninguna llamada cruzada. auth-service y core-service tienen
// cada uno su propia copia de este modelo — ver leyenda al inicio del archivo.
model AuditLog {
  id          BigInt  @id @default(autoincrement())
  tableName   String
  recordId    String
  action      String
  before      Json?
  after       Json?
  performedBy String // desde JWT userId — sin llamada cruzada
  requestId   String?

  createdAt DateTime @default(now())

  @@map("audit_log")
}

// SERVICE: core-service (puerto 3002)
// Dueño de: marcas, redes sociales, campañas y su equipo, publicaciones,
// su máquina de estados y fan-out multi-red, medios, métricas simuladas,
// score digital, reportes (PDF/CSV) e ideas de contenido (Alexa Skill).
// ══════════════════════════════════════════════════════════════════════
//
// Esta sección fusiona lo que antes eran 3 secciones separadas
// (brands-service, content-service, analytics-service). Todas las FKs que
// antes cruzaban entre ellas (Post.brandId, Post.campaignId,
// PostSocialAccount.socialAccountId, PostMetric.postSocialAccountId,
// BrandScore.brandId, Report.brandId, ContentIdea.campaignId, el cron de
// analytics leyendo SocialNetwork.baseEngagementRate, etc.) dejan de ser
// "FK tonta + llamada HTTP" y pasan a ser `@relation` normales de Prisma:
// ahora es el mismo proceso y la misma base de datos. Esto también resuelve
// de raíz la violación que hoy existe en el código real
// (score.service.ts / metrics-cron.service.ts leían Post/BrandProfile
// cruzando límites de servicio) — bajo este modelo esa lectura es local y
// legítima. Por la misma razón, los snapshots denormalizados que existían
// solo para evitar HTTP entre servicios (brandName/campaignName en Post,
// socialAccountHandle/socialNetworkCode en PostSocialAccount) también se
// retiran aquí: ya no hay HTTP que evitar, un join normal basta.
//
// Lo único que sigue siendo una FK cruzada real es hacia auth-service
// (identidad: ownerId, cmId, createdBy, changedBy, uploadedBy,
// requestedBy, ContentIdea.createdBy → User). Category, Specialty y
// SocialNetwork ya son catálogos locales de core-service.
//
// Nota histórica: antes había que decidir si Campaign vivía en
// brands-service o content-service (ver docs/architecture.md vs.
// docs/database.md). Con la fusión en core-service la pregunta queda sin
// objeto — ambos son el mismo servicio — se deja documentada en
// docs/base/service-boundaries.md solo como antecedente.

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL_CORE")
}

// ── Catálogos ─────────────────────────────────────────────────────────

// Áreas temáticas: clasifican a usuarios operativos (CM/Diseñador),
// campañas (core-service) y marcas (core-service). Es el eje del
// sistema de matching (orientativo, no restrictivo — regla de negocio #8).
//
// NOTA — llamada de juicio, no hecho cerrado: Category se queda aquí en
// auth-service junto a Role/Module/Action, reutilizando una dependencia
// que casi todo servicio ya necesita hacia auth. La alternativa
// (core-service, mayor consumidor real vía Brand+Campaign) es igual de
// defendible — ver docs/base/service-boundaries.md. Cambiarlo es mover
// esta sección, no rediseñar el archivo.
model Category {
  id   String @id @default(uuid())
  name String @unique

  userCategories UserCategory[] // uso local (mismo servicio)

  // Consumido cruzado (sin @relation posible — otra BD):
  // core-service.Campaign.categoryId, core-service.Brand.categoryId
  // → FK → auth-service.Category.id (sin constraint; resolver vía HTTP si
  //   se necesita el nombre, aunque en la mayoría de vistas basta el id)

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@map("categories")
}

// Habilidades técnicas exclusivas de CM y Diseñador. NO aplica a marcas ni
// campañas — por eso no tiene ningún consumidor cruzado, a diferencia de
// Category.
model Specialty {

  name String @unique

  userSpecialties UserSpecialty[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@map("specialties")
}

}

// Estado único de la publicación: cubre tanto el flujo de aprobación
// (borrador → en_revision → aprobado/rechazado) como el de publicación
// multi-red (programado → publicando → publicado/parcial/error/cancelado).
// post_status_history registra cada transición (actor, fecha, comentario).
enum PostStatus {
  borrador
  en_revision
  aprobado
  rechazado
  programado
  publicando
  publicado
  parcial
  error
  cancelado
}

enum PostSocialAccountStatus {
  pendiente
  publicando
  publicado
  error
  cancelado
}

// Capa 1 — La intención del usuario (independiente de la red)
model Post {
  id         String    @id @default(uuid())
  brandId    String
  brand      Brand     @relation(fields: [brandId], references: [id])
  campaignId String?
  campaign   Campaign? @relation(fields: [campaignId], references: [id])

  content     String     @db.Text
  status      PostStatus @default(borrador)
  scheduledAt DateTime?
  publishedAt DateTime?

  createdBy String // FK → auth-service.User.id (CM o Diseñador; sin
  // constraint, viene del JWT al escribir)

  // null hasta que Ayrshare acepta el post
  ayrsharePostId String? @unique

  socialAccounts PostSocialAccount[]
  statusHistory  PostStatusHistory[]
  media          PostMedia[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@map("posts")
}

// Inmutable — historial de transiciones de estado del Post
model PostStatusHistory {
  id         BigInt      @id @default(autoincrement())
  postId     String
  post       Post        @relation(fields: [postId], references: [id])
  fromStatus PostStatus?
  toStatus   PostStatus
  changedBy  String // FK → auth-service.User.id (sin constraint; JWT al escribir)
  comment    String?     @db.Text // obligatorio si toStatus = rechazado o cancelado

  createdAt DateTime @default(now())

  @@map("post_status_history")
}

// Capa 2 — Publicación física por red social

  publishedAt  DateTime?
  errorMessage String?

  updatedAt DateTime @updatedAt

  @@unique([postId, socialAccountId])
  @@map("post_social_accounts")
}

// ── Ideas de contenido ──────────────────────────────────────────────────
// Capturadas por la Alexa Skill (SaveIdeaIntent/SaveCustomIdeaIntent, vía
// alexa-service, que no tiene BD propia y llama al endpoint de este
// servicio) y, potencialmente, desde el panel web más adelante — mismo
// modelo, sin distinción de canal de origen a nivel de dato (no hay razón
// funcional para que una idea "sepa" si nació por voz o por texto, solo si
// es sugerida por IA o dictada por el usuario — ver `source`). Dueño:
// core-service, porque una idea de contenido es contenido, misma familia
// que Post/Media.
enum ContentIdeaSource {
  sugerida // generada por GenerateContentIdeasIntent y guardada tal cual
  propia   // dictada libremente por el usuario (SaveCustomIdeaIntent)
}

model ContentIdea {
  id         String   @id @default(uuid())
  campaignId String
  campaign   Campaign @relation(fields: [campaignId], references: [id])

  createdBy String // FK → auth-service.User.id (sin constraint de BD; viene
  // del JWT al escribir, sin llamada — mismo patrón que Post.createdBy)

  title  String? // null cuando source = propia y el usuario no dictó título
  text   String  @db.Text
  source ContentIdeaSource @default(sugerida)

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@map("content_ideas")
}

// Endpoints que expone core-service para este modelo (consumidos por
// alexa-service, nunca directamente por el Lambda):
//   POST   /campaigns/:campaignId/ideas   (SaveIdeaIntent, SaveCustomIdeaIntent)
//   GET    /campaigns/:campaignId/ideas   (GetSavedIdeasIntent)
//   DELETE /ideas/:id                     (DeleteIdeaIntent, resuelto por título
//          hablado — búsqueda por title/text dentro del campaignId de sesión)
// Requieren JwtAuthGuard + BrandAccessGuard + PermissionGuard como cualquier
// otro endpoint de negocio del proyecto (.agents/backend.md).

// Capa 3 — Snapshot de métricas por red (cron job).
model PostMetric {
  id                  String            @id @default(uuid())
  postSocialAccountId String
  postSocialAccount   PostSocialAccount @relation(fields: [postSocialAccountId], references: [id])

  likes      Int?
  comments   Int?
  shares     Int?
  views      Int?
  reach      Int?
  engagement Float? // tasa calculada: (likes+comments+shares) / reach

  capturedAt DateTime @default(now())

  @@map("post_metrics")
}

// Score = (Consistencia×0.30) + (Engagement×0.40) + (Frecuencia×0.30) — ver
// CLAUDE.md sobre la discrepancia entre la fórmula documentada (3 factores)
// y la implementada hoy en score.service.ts (4 factores, incluye
// Cobertura×0.20) — confirmar con el usuario cuál es la vigente antes de
// tocar el cálculo real.
// `coverage` es informativa: NO entra al cálculo ponderado del score.
model BrandScore {
  id      String @id @default(uuid())
  brandId String
  brand   Brand  @relation(fields: [brandId], references: [id])

  score          Float
  consistency    Float
  engagement     Float
  frequency      Float
  coverage       Float
  classification String
  snapshotDate   DateTime @default(now())

  @@map("brand_scores")
}
